CREATE DATABASE tomatix;

USE tomatix;

CREATE TABLE empresa (
	id INT PRIMARY KEY AUTO_INCREMENT,
	razaoSocial VARCHAR(50) NOT NULL,
	CNPJ CHAR(14) NOT NULL UNIQUE, 
	dtCadastro DATETIME NOT NULL
);

CREATE TABLE usuario (
	id INT PRIMARY KEY AUTO_INCREMENT,
	nome VARCHAR(50) NOT NULL,
	login VARCHAR(50) NOT NULL UNIQUE,
	email VARCHAR(50) NOT NULL UNIQUE,
	senha VARCHAR(50) NOT NULL,
	dtCadastro DATETIME DEFAULT CURRENT_TIMESTAMP(),
	permissao VARCHAR(50) NOT NULL,
	CONSTRAINT chkPermissao
	CHECK(permissao IN('admin', 'funcionario')),
	fkEmpresa INT NOT NULL,
	CONSTRAINT fkUsuarioEmpresa FOREIGN KEY (fkEmpresa) REFERENCES empresa(id)
);

CREATE TABLE estufa (
	id INT PRIMARY KEY AUTO_INCREMENT,
	dtCadastro DATETIME NOT NULL,
	area DECIMAL(10,2) NOT NULL, 
	CONSTRAINT chkArea
	CHECK (area > 0),
	endereco VARCHAR(100) NOT NULL,
	cidade VARCHAR(100) NOT NULL,
	estado VARCHAR(100) NOT NULL,
	dtPlantio DATETIME,
	status VARCHAR(7) NOT NULL DEFAULT 'ativo',
	CONSTRAINT chkStatus
	CHECK(status IN('ativo', 'inativo')),
	fkEmpresa INT NOT NULL,
	CONSTRAINT fkEstufaEmpresa FOREIGN KEY (fkEmpresa) REFERENCES empresa(id)
);

CREATE TABLE sensor (
	id INT PRIMARY KEY AUTO_INCREMENT,
	dtInstalacao DATETIME NOT NULL,
	nSerie VARCHAR(50) NOT NULL,
	unMedida VARCHAR(20) NOT NULL,
	status_sensor VARCHAR(50),
	CONSTRAINT chkStatusSensor
	CHECK (status_sensor IN('ativo', 'inativo')),
	idEstufa INT NOT NULL,
	CONSTRAINT fkSensorEstufa FOREIGN KEY (idEstufa) REFERENCES estufa(id)
);

CREATE TABLE leituraLuminosidade (
	id INT PRIMARY KEY AUTO_INCREMENT,
	dtHora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP(),
	lux DECIMAL(10,2) NOT NULL,
	ppfd DECIMAL (10,2) NOT NULL,
	dli DECIMAL (5,2) NOT NULL,
	idSensor INT NOT NULL,
	CONSTRAINT fkLeituraSensor FOREIGN KEY (idSensor) REFERENCES sensor(id)
);
